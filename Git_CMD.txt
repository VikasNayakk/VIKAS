# ============================================================
# Git Command Master List (Windows CMD Friendly)
# ============================================================

# ------------------------------------------------------------
# 0) Basic Setup (Run once on new machine)
# ------------------------------------------------------------
git --version
git config --global user.name "Your Name"
git config --global user.email "you@example.com"
git config --global init.defaultBranch main
git config --global core.editor "notepad"
git config --global pull.rebase false
git config --list

# ------------------------------------------------------------
# 1) SSH Setup (GitHub)
# ------------------------------------------------------------
ssh-keygen -t rsa -b 4096 -C "you@example.com"
type %USERPROFILE%\.ssh\id_rsa.pub
notepad %USERPROFILE%\.ssh\id_rsa.pub
ssh -T git@github.com

# ------------------------------------------------------------
# 2) Create / Clone / Init
# ------------------------------------------------------------
git clone https://github.com/OWNER/REPO.git
git clone -b main git@github.com:OWNER/REPO.git
git init
git remote add origin https://github.com/OWNER/REPO.git
git remote add origin git@github.com:OWNER/REPO.git
git remote -v

# ------------------------------------------------------------
# 3) Change Remote URL (HTTPS <-> SSH)
# ------------------------------------------------------------
git remote set-url origin git@github.com:OWNER/REPO.git
git remote set-url origin https://github.com/OWNER/REPO.git

# ------------------------------------------------------------
# 4) Daily Workflow (Status / Add / Commit / Push / Pull)
# ------------------------------------------------------------
git status
git add .
git add path\to\file.txt
git commit -m "Message"
git push origin main
git pull origin main

# ------------------------------------------------------------
# 5) Branching
# ------------------------------------------------------------
git branch
git branch new-branch
git checkout new-branch
git checkout -b feature/login
git branch -d old-branch
git push origin feature/login
git pull origin feature/login

# ------------------------------------------------------------
# 6) Merge / Rebase
# ------------------------------------------------------------
git checkout main
git merge feature/login
git rebase main
git rebase --abort
git rebase --continue

# ------------------------------------------------------------
# 7) Log / History
# ------------------------------------------------------------
git log
git log --oneline --graph --decorate
git show
git diff
git diff --staged

# ------------------------------------------------------------
# 8) Undo / Fix
# ------------------------------------------------------------
git restore path\to\file.txt
git restore --staged path\to\file.txt
git checkout -- path\to\file.txt
git reset --soft HEAD~1
git reset --mixed HEAD~1
git reset --hard HEAD~1
git revert HEAD

# ------------------------------------------------------------
# 9) Stash
# ------------------------------------------------------------
git stash
git stash list
git stash apply
git stash pop
git stash drop

# ------------------------------------------------------------
# 10) Tags (Releases)
# ------------------------------------------------------------
git tag
git tag v1.0.0
git tag -a v1.0.0 -m "Release 1.0.0"
git push origin v1.0.0
git push origin --tags
git tag -d v1.0.0

# ------------------------------------------------------------
# 11) Fetch / Clean
# ------------------------------------------------------------
git fetch
git fetch --all
git remote prune origin
git clean -fd

# ------------------------------------------------------------
# 12) Submodules
# ------------------------------------------------------------
git submodule add https://github.com/OWNER/REPO.git path\to\module
git submodule init
git submodule update
git submodule update --remote

# ------------------------------------------------------------
# 13) Helpful Shortcuts (Aliases)
# ------------------------------------------------------------
git config --global alias.st status
git config --global alias.co checkout
git config --global alias.br branch
git config --global alias.cm "commit -m"
git config --global alias.lg "log --oneline --graph --decorate"

# ------------------------------------------------------------
# 14) Example Windows CMD One-Liners
# ------------------------------------------------------------
cd C:\Aliens\Project && git status
cd C:\Aliens\Project && git add . && git commit -m "Update" && git push origin main
cd C:\Aliens\Project && git pull origin main
